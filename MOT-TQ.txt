var color C_NAVY   = #1A2540
var color C_GREEN  = #15803D
var color C_RED    = #CC2936
var color C_WHITE  = #FFFFFF
var color C_GREY   = #888888
var color C_CYAN   = #00BCD4
var color C_PURPLE = #9C27B0

var color C_TEAL   = #009688
var color C_PINK   = #E91E63
var color C_LIME   = #CDDC39

// ─────────────────────────────────────────────────────────────
// LINE STYLE CONVERSION FUNCTION
// ─────────────────────────────────────────────────────────────
getLineStyle(styleName) =>
    if styleName == "Dotted"
        line.style_dotted
    else if styleName == "Dashed"
        line.style_dashed
    else
        line.style_solid

// ─────────────────────────────────────────────────────────────
// INPUTS
// ─────────────────────────────────────────────────────────────

// === MOVING AVERAGE OVERLAYS ===
var string G_MA = "=== MOVING AVERAGE OVERLAYS ==="
showSMA9   = input.bool(false, title="Show SMA 9", group=G_MA)
sma9Color  = input.color(C_BLUE, title="SMA 9 Color", group=G_MA)
sma9Width  = input.int(1, title="SMA 9 Width", minval=1, maxval=5, group=G_MA)

showEMA20  = input.bool(false, title="Show EMA 20", group=G_MA)
ema20Color = input.color(C_TEAL, title="EMA 20 Color", group=G_MA)
ema20Width = input.int(1, title="EMA 20 Width", minval=1, maxval=5, group=G_MA)

ema20      = ta.ema(close, 20)
sma9       = ta.sma(close, 9)
ema50      = ta.ema(close, 50)
ema200     = ta.ema(close, 200)

// === PRIOR DAY HIGH/LOW ===
var string G_PD = "=== PRIOR DAY HIGH/LOW ==="
showPriorDay = input.bool(true, title="Show Prior Day High/Low", group=G_PD)
priorDayColor = input.color(C_CYAN, title="Prior Day Line Color", group=G_PD)
priorDayWidth = input.int(1, title="Prior Day Line Width", minval=1, maxval=3, group=G_PD)
priorDayStyle = input.string("Dotted", title="Prior Day Line Style", options=["Solid", "Dotted", "Dashed"], group=G_PD)
showPriorDayLabels = input.bool(true, title="Show Prior Day Labels", group=G_PD)

// === OVERNIGHT RANGE ===
var string G_ON = "=== OVERNIGHT RANGE ==="
showOvernight = input.bool(true, title="Show Overnight Range", group=G_ON)
overnightColor = input.color(C_PURPLE, title="Overnight Line Color", group=G_ON)
overnightWidth = input.int(1, title="Overnight Line Width", minval=1, maxval=3, group=G_ON)
overnightStyle = input.string("Dashed", title="Overnight Line Style", options=["Solid", "Dotted", "Dashed"], group=G_ON)
showOvernightMid = input.bool(true, title="Show Overnight Midpoint", group=G_ON)
showOvernightLabels = input.bool(true, title="Show Overnight Labels", group=G_ON)

// === POINT OF CONTROL & VALUE AREA ===
var string G_POC = "=== POINT OF CONTROL ==="
showPOC = input.bool(true, title="Show Point of Control", group=G_POC)
pocColor = input.color(C_ORANGE, title="POC Color", group=G_POC)
pocWidth = input.int(1, title="POC Line Width", minval=1, maxval=3, group=G_POC)
showPOCLabel = input.bool(true, title="Show POC Label", group=G_POC)

showVA = input.bool(true, title="Show Value Area", group=G_POC)
vaColor = input.color(C_ORANGE, title="Value Area Color", group=G_POC)
vaWidth = input.int(1, title="Value Area Line Width", minval=1, maxval=2, group=G_POC)
vaStyle = input.string("Dotted", title="Value Area Style", options=["Solid", "Dotted", "Dashed"], group=G_POC)

// === SESSION POC ===
var string G_SPOC = "=== SESSION POC ==="
showSessionPOC = input.bool(true, title="Show Session POC", group=G_SPOC)
sessionPOCColor = input.color(C_GOLD, title="Session POC Color", group=G_SPOC)
sessionPOCWidth = input.int(1, title="Session POC Width", minval=1, maxval=2, group=G_SPOC)

// === OTHER INDICATOR SETTINGS ===
volLen     = input.int(20,    title="Volume MA Length")
volMulti   = input.float(1.0, title="Min Volume Multiplier vs MA", step=0.1)
volMA      = ta.sma(volume, volLen)

srLen      = input.int(20, title="SR Swing Length")
priorHigh  = ta.highest(high, srLen)[1]
priorLow   = ta.lowest(low,   srLen)[1]
buffer     = input.float(0.002, title="SR Buffer (% of price)", step=0.001)

sess1      = input.session("0830-1100", title="Primary Session")
sess2      = input.session("1300-1500", title="Secondary Session")

bodyRatio  = input.float(0.5, title="Min Body-to-Range Ratio", step=0.05)

atrLen     = input.int(14,    title="ATR Length")
maxAtrMult = input.float(1.5, title="Max Stop = N x ATR", step=0.25)

minRR      = input.float(2.0, title="Min Risk-Reward Ratio", step=0.25)

htfTF      = input.timeframe("D", title="Higher Timeframe")

showLabel  = input.bool(true, title="Show signal label on chart")
showScore  = input.bool(true, title="Show score in label")
showBg     = input.bool(true, title="Highlight bars when gate open")

// TABLE POSITION CONTROL
tablePos   = input.string("Top Right", title="Table Position", options=["Top Right", "Top Left", "Bottom Right", "Bottom Left"])

// ─────────────────────────────────────────────────────────────
// DERIVED VALUES
// ─────────────────────────────────────────────────────────────
atrVal     = ta.atr(atrLen)
barRange   = high - low
bodySize   = math.abs(close - open)
strongBar  = barRange > 0 and (bodySize / barRange) >= bodyRatio

[htfClose, htfEMAval] = request.security(syminfo.tickerid, htfTF, [close, ta.ema(close, 50)], lookahead=barmerge.lookahead_off)

stopL      = close - priorLow
stopS      = priorHigh - close
targetL    = priorHigh - close
targetS    = close - priorLow
rrL        = stopL > 0 ? targetL / stopL : 0.0
rrS        = stopS > 0 ? targetS / stopS : 0.0

inSession1 = not na(time(timeframe.period, sess1))
inSession2 = not na(time(timeframe.period, sess2))
inSession  = inSession1 or inSession2

volQualified = volume >= volMA * volMulti
sessionQualified = inSession

// ─────────────────────────────────────────────────────────────
// INSTITUTIONAL LEVELS
// ─────────────────────────────────────────────────────────────
isNewDay = ta.change(time("D")) != 0

var float priorDayHigh = na
var float priorDayLow = na
var float currentDayHigh = high
var float currentDayLow = low

if isNewDay
    priorDayHigh := currentDayHigh[1]
    priorDayLow := currentDayLow[1]
    currentDayHigh := high
    currentDayLow := low
else
    currentDayHigh := math.max(currentDayHigh, high)
    currentDayLow := math.min(currentDayLow, low)

var float overnightHigh = na
var float overnightLow = na
var float overnightMid = na

if isNewDay
    overnightHigh := high
    overnightLow := low
    overnightMid := (high + low) / 2

barsSinceOpen = ta.barssince(isNewDay)
if barsSinceOpen <= 12
    overnightHigh := math.max(overnightHigh, high)
    overnightLow := math.min(overnightLow, low)
    overnightMid := (overnightHigh + overnightLow) / 2

// POC calculation
pocBarsBack = 100
float highestPrice = ta.highest(high, pocBarsBack)
float lowestPrice = ta.lowest(low, pocBarsBack)
float pocRaw = (highestPrice + lowestPrice) / 2
float vahRaw = highestPrice - (highestPrice - lowestPrice) * 0.3
float valRaw = lowestPrice + (highestPrice - lowestPrice) * 0.3

var float pocPrice = na
var float valueAreaHigh = na
var float valueAreaLow = na

if bar_index % 50 == 0
    pocPrice := pocRaw
    valueAreaHigh := vahRaw
    valueAreaLow := valRaw

// Session POC
var float sessionPOC = na
var float sessionHigh = high
var float sessionLow = low

if isNewDay
    sessionHigh := high
    sessionLow := low
else
    sessionHigh := math.max(sessionHigh, high)
    sessionLow := math.min(sessionLow, low)
    sessionPOC := (sessionHigh + sessionLow) / 2

// ─────────────────────────────────────────────────────────────
// EIGHT CHECKS — LONG
// ─────────────────────────────────────────────────────────────
l1 = ema20 > ema50 and ema50 > ema200 and close > ema20
l2 = volQualified
l3 = math.abs(close - priorLow) / close <= buffer
l4 = sessionQualified
l5 = close > open and strongBar
l6 = stopL > 0 and stopL <= atrVal * maxAtrMult
l7 = rrL >= minRR
l8 = htfClose > htfEMAval

scoreL = (l1 ? 1 : 0) + (l2 ? 1 : 0) + (l3 ? 1 : 0) + (l4 ? 1 : 0) + (l5 ? 1 : 0) + (l6 ? 1 : 0) + (l7 ? 1 : 0) + (l8 ? 1 : 0)

// ─────────────────────────────────────────────────────────────
// EIGHT CHECKS — SHORT
// ─────────────────────────────────────────────────────────────
s1 = ema20 < ema50 and ema50 < ema200 and close < ema20
s2 = volQualified
s3 = math.abs(close - priorHigh) / close <= buffer
s4 = sessionQualified
s5 = close < open and strongBar
s6 = stopS > 0 and stopS <= atrVal * maxAtrMult
s7 = rrS >= minRR
s8 = htfClose < htfEMAval

scoreS = (s1 ? 1 : 0) + (s2 ? 1 : 0) + (s3 ? 1 : 0) + (s4 ? 1 : 0) + (s5 ? 1 : 0) + (s6 ? 1 : 0) + (s7 ? 1 : 0) + (s8 ? 1 : 0)

// ─────────────────────────────────────────────────────────────
// GATE
// ─────────────────────────────────────────────────────────────
gateL    = scoreL >= 6
gateS    = scoreS >= 6
gateNewL = gateL and not gateL[1]
gateNewS = gateS and not gateS[1]

// ─────────────────────────────────────────────────────────────
// SIGNAL LABELS
// ─────────────────────────────────────────────────────────────
scoreTxtL = showScore ? "\n" + str.tostring(scoreL) + "/8" : ""
scoreTxtS = showScore ? "\n" + str.tostring(scoreS) + "/8" : ""

if gateNewL and showLabel
    label.new(bar_index, low - atrVal * 0.5, text="▲ FIRE" + scoreTxtL, style=label.style_label_up, color=C_GREEN, textcolor=C_WHITE, size=size.normal)

if gateNewS and showLabel
    label.new(bar_index, high + atrVal * 0.5, text="▼ FIRE" + scoreTxtS, style=label.style_label_down, color=C_RED, textcolor=C_WHITE, size=size.normal)

// ─────────────────────────────────────────────────────────────
// MOVING AVERAGE PLOTS
// ─────────────────────────────────────────────────────────────
plot(showSMA9 ? sma9 : na, title="SMA 9", color=sma9Color, linewidth=sma9Width, style=plot.style_line)
plot(showEMA20 ? ema20 : na, title="EMA 20", color=ema20Color, linewidth=ema20Width, style=plot.style_line)

// ─────────────────────────────────────────────────────────────
// INSTITUTIONAL LEVELS VISUALS
// ─────────────────────────────────────────────────────────────

// Prior Day High/Low
if showPriorDay and not na(priorDayHigh)
    lineStyle = getLineStyle(priorDayStyle)
    line.new(bar_index - 1, priorDayHigh, bar_index, priorDayHigh, color=priorDayColor, style=lineStyle, width=priorDayWidth)
    line.new(bar_index - 1, priorDayLow, bar_index, priorDayLow, color=priorDayColor, style=lineStyle, width=priorDayWidth)
    if showPriorDayLabels
        label.new(bar_index, priorDayHigh, text="Prior High", style=label.style_label_left, color=priorDayColor, textcolor=C_WHITE, size=size.tiny)
        label.new(bar_index, priorDayLow, text="Prior Low", style=label.style_label_left, color=priorDayColor, textcolor=C_WHITE, size=size.tiny)

// Overnight Range
if showOvernight and not na(overnightHigh) and not na(overnightLow)
    lineStyle = getLineStyle(overnightStyle)
    line.new(bar_index - 1, overnightHigh, bar_index, overnightHigh, color=overnightColor, style=lineStyle, width=overnightWidth)
    line.new(bar_index - 1, overnightLow, bar_index, overnightLow, color=overnightColor, style=lineStyle, width=overnightWidth)
    
    if showOvernightMid and not na(overnightMid)
        line.new(bar_index - 1, overnightMid, bar_index, overnightMid, color=color.new(overnightColor, 60), style=line.style_dotted, width=1)
    
    if showOvernightLabels
        label.new(bar_index, overnightHigh, text="ON High", style=label.style_label_left, color=overnightColor, textcolor=C_WHITE, size=size.tiny)
        label.new(bar_index, overnightLow, text="ON Low", style=label.style_label_left, color=overnightColor, textcolor=C_WHITE, size=size.tiny)
        if showOvernightMid
            label.new(bar_index, overnightMid, text="ON Mid", style=label.style_label_left, color=color.new(overnightColor, 60), textcolor=C_WHITE, size=size.tiny)

// Point of Control
if showPOC and not na(pocPrice)
    line.new(bar_index - 1, pocPrice, bar_index, pocPrice, color=pocColor, style=line.style_solid, width=pocWidth)
    if showPOCLabel
        label.new(bar_index, pocPrice, text="POC", style=label.style_label_left, color=pocColor, textcolor=C_WHITE, size=size.small)

// Value Area
if showVA and not na(valueAreaHigh) and not na(valueAreaLow)
    vaLineStyle = getLineStyle(vaStyle)
    line.new(bar_index - 1, valueAreaHigh, bar_index, valueAreaHigh, color=vaColor, style=vaLineStyle, width=vaWidth)
    line.new(bar_index - 1, valueAreaLow, bar_index, valueAreaLow, color=vaColor, style=vaLineStyle, width=vaWidth)

// Session POC
if showSessionPOC and not na(sessionPOC)
    line.new(bar_index - 1, sessionPOC, bar_index, sessionPOC, color=sessionPOCColor, style=line.style_dotted, width=sessionPOCWidth)

// ─────────────────────────────────────────────────────────────
// BACKGROUND TINT
// ─────────────────────────────────────────────────────────────
bgColor = gateL and showBg ? color.new(C_GREEN, 92) : gateS and showBg ? color.new(C_RED, 92) : na
bgcolor(bgColor)

// ─────────────────────────────────────────────────────────────
// TABLE POSITION CONVERSION
// ─────────────────────────────────────────────────────────────
getTablePosition(pos) =>
    if pos == "Top Right"
        position.top_right
    else if pos == "Top Left"
        position.top_left
    else if pos == "Bottom Right"
        position.bottom_right
    else
        position.bottom_left

// ─────────────────────────────────────────────────────────────
// TABLE
// ─────────────────────────────────────────────────────────────
f_row(int row, string num, string lbl, bool passL, bool passS, table tbl) =>
    bgL = passL ? color.new(C_GREEN, 65) : color.new(C_RED, 65)
    bgS = passS ? color.new(C_GREEN, 65) : color.new(C_RED, 65)
    table.cell(tbl, 0, row, num, text_color=C_GOLD, text_size=size.small, bgcolor=color.new(C_NAVY, 10))
    table.cell(tbl, 1, row, lbl, text_color=C_WHITE, text_size=size.small, bgcolor=color.new(C_NAVY, 10))
    table.cell(tbl, 2, row, passL ? "L ✔" : "L ✘", text_color=C_WHITE, text_size=size.small, bgcolor=bgL)
    table.cell(tbl, 3, row, passS ? "S ✔" : "S ✘", text_color=C_WHITE, text_size=size.small, bgcolor=bgS)

var table scorecard = table.new(getTablePosition(tablePos), 4, 12, bgcolor=color.new(C_NAVY, 5), border_color=C_GOLD, border_width=1, frame_color=C_GOLD, frame_width=2)

gateLTxt = gateL ? "▲ FIRE " + str.tostring(scoreL) + "/8" : "▲ " + str.tostring(scoreL) + "/8"
gateSTxt = gateS ? "▼ FIRE " + str.tostring(scoreS) + "/8" : "▼ " + str.tostring(scoreS) + "/8"

table.cell(scorecard, 0, 0, "MOT · TQ", text_color=C_GOLD, text_size=size.small, bgcolor=C_NAVY)
table.cell(scorecard, 1, 0, "Check", text_color=C_GOLD, text_size=size.small, bgcolor=C_NAVY)
table.cell(scorecard, 2, 0, gateLTxt, text_color=gateL ? C_GREEN : C_GREY, text_size=size.small, bgcolor=C_NAVY)
table.cell(scorecard, 3, 0, gateSTxt, text_color=gateS ? C_RED : C_GREY, text_size=size.small, bgcolor=C_NAVY)

f_row(1, "①", "Trend", l1, s1, scorecard)
f_row(2, "②", "Liquidity", l2, s2, scorecard)
f_row(3, "③", "Location", l3, s3, scorecard)
f_row(4, "④", "Timing", l4, s4, scorecard)
f_row(5, "⑤", "Confirmation", l5, s5, scorecard)
f_row(6, "⑥", "Risk", l6, s6, scorecard)
f_row(7, "⑦", "Reward", l7, s7, scorecard)
f_row(8, "⑧", "Context", l8, s8, scorecard)

table.cell(scorecard, 0, 10, "GATE", text_color=C_GOLD, text_size=size.small, bgcolor=C_NAVY)
table.cell(scorecard, 1, 10, ">= 6/8", text_color=C_GOLD, text_size=size.small, bgcolor=C_NAVY)
table.cell(scorecard, 2, 10, gateL ? "OPEN" : "CLOSED", text_color=gateL ? C_GREEN : C_GREY, text_size=size.small, bgcolor=C_NAVY)
table.cell(scorecard, 3, 10, gateS ? "OPEN" : "CLOSED", text_color=gateS ? C_RED : C_GREY, text_size=size.small, bgcolor=C_NAVY)

table.cell(scorecard, 0, 11, "R:R", text_color=C_GREY, text_size=size.tiny, bgcolor=color.new(C_NAVY, 20))
table.cell(scorecard, 1, 11, "ratio", text_color=C_GREY, text_size=size.tiny, bgcolor=color.new(C_NAVY, 20))
table.cell(scorecard, 2, 11, str.tostring(math.round(rrL, 2), "#.##"), text_color=rrL >= minRR ? C_GREEN : C_GREY, text_size=size.tiny, bgcolor=color.new(C_NAVY, 20))
table.cell(scorecard, 3, 11, str.tostring(math.round(rrS, 2), "#.##"), text_color=rrS >= minRR ? C_GREEN : C_GREY, text_size=size.tiny, bgcolor=color.new(C_NAVY, 20))

// ─────────────────────────────────────────────────────────────
// ALERTS
// ─────────────────────────────────────────────────────────────
alertcondition(gateNewL, title="LONG Gate Passed", message="LONG setup qualifies (>=6/8)")
alertcondition(gateNewS, title="SHORT Gate Passed", message="SHORT setup qualifies (>=6/8)")

// ─────────────────────────────────────────────────────────────
// PLOTS
// ─────────────────────────────────────────────────────────────
plot(scoreL, title="Long Score", display=display.data_window + display.status_line, color=C_GREEN)
plot(scoreS, title="Short Score", display=display.data_window + display.status_line, color=C_RED)

// ──────────────────────────────────────────────────────────────
// INPUTS — BAR NUMBERS
// ──────────────────────────────────────────────────────────────
var g_STYLE = "Display & Style"
float labelOffset = input.float(0.5, "Label & Bar Number Offset (x ATR)", minval=0, step=0.1, group=g_STYLE)
float _atr14 = ta.atr(14)
float _labelOff = labelOffset * nz(_atr14, 1.0)
// ══════════════════════════════════════════════════════════════
// SECTION 4: TIME-OF-DAY CONTEXT
// ══════════════════════════════════════════════════════════════

int nyHour = hour(time, "America/New_York")
int nyMinute = minute(time, "America/New_York")
int nyTime = nyHour * 100 + nyMinute

int todScore = nyTime >= 1030 and nyTime < 1130 ? 1 : nyTime >= 1400 and nyTime < 1500 ? 1 : nyTime >= 1130 and nyTime < 1330 ? -1 : nyTime >= 1000 and nyTime < 1030 ? -1 : 0

bool isLunchChop = nyTime >= 1130 and nyTime < 1330

var g_BARNUM = "Bar Numbers"
bool showBarNums = input.bool(true, "Show Bar Numbers", group=g_BARNUM)
int barNumFreq = input.int(1, "Show Every Nth Bar", minval=1, maxval=10, group=g_BARNUM, tooltip="1 = every bar, 2 = every other bar, etc.")
string barNumPos = input.string("Below Bar", "Label Position", options=["Below Bar","Above Bar"], group=g_BARNUM)
string barNumSize = input.string("Tiny", "Label Size", options=["Tiny","Small","Normal"], group=g_BARNUM)
color barNumRTHColor = input.color(color.new(color.white, 30), "RTH Number Color", group=g_BARNUM)
color barNumETHColor = input.color(color.new(color.gray, 30), "ETH Number Color", group=g_BARNUM)
string barNumScope = input.string("RTH + ETH", "Show Numbers For", options=["RTH + ETH","RTH Only","ETH Only"], group=g_BARNUM, tooltip="Choose whether to display bar numbers during RTH, ETH, or both.")


var int barCount = 0
var bool wasRTHBar = false

bool isRTHBar = nyTime >= 930 and nyTime < 1600

if isRTHBar and not wasRTHBar
    barCount := 1
    wasRTHBar := true
else if not isRTHBar and wasRTHBar
    barCount := 1
    wasRTHBar := false
else
    barCount += 1

_barNumSize = switch barNumSize
    "Tiny"   => size.tiny
    "Small"  => size.small
    "Normal" => size.normal
    =>         size.tiny

bool showThisBar = showBarNums
showThisBar := showThisBar and (barCount % barNumFreq == 0 or barCount == 1)

if barNumScope == "RTH Only" and not isRTHBar
    showThisBar := false
if barNumScope == "ETH Only" and isRTHBar
    showThisBar := false

string barNumText = (isRTHBar ? "" : "E") + str.tostring(barCount)
color barNumClr = isRTHBar ? barNumRTHColor : barNumETHColor

if showThisBar
    string _barNumStyle = barNumPos == "Below Bar" ? label.style_label_up : label.style_label_down
    float _barNumY = barNumPos == "Below Bar" ? low - _labelOff : high + _labelOff
    label.new(bar_index, _barNumY, barNumText, style=_barNumStyle, color=color.new(color.black, 100), textcolor=barNumClr, size=_barNumSize)
